    Rel(telegram, estudiante, "Entrega avisos")
    Rel(estudiante, taia, "Consulta tareas mediante la aplicación móvil")
    Rel(taia, telegram, "Envía notificaciones")

    UpdateElementStyle(estudiante, $bgColor="#08427b", $fontColor="#ffffff", $borderColor="#052e56")

    UpdateElementStyle(LLM, $bgColor="#999999", $fontColor="#ffffff", $borderColor="#6b6b6b")

    UpdateRelStyle(telegram, estudiante, $offsetX="-40", $offsetY="-50")
