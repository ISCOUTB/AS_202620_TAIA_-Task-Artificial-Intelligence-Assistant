El siguiente diagrama representa la **línea base backend actualmente ejecutable**. La aplicación móvil y PostgreSQL permanecen como componentes de la arquitectura objetivo y no se presentan como implementaciones disponibles en esta versión.

```mermaid
C4Container
    title Diagrama de Contenedores - TAIA (línea base ejecutable)

    Person(estudiante, "Estudiante", "Se autentica, registra y consulta tareas y gestiona recordatorios.")
    System_Ext(telegram, "Telegram", "Canal externo utilizado para vinculación y entrega de notificaciones.")
    System_Ext(llm, "Proveedor LLM", "Proveedor externo para interpretar mensajes en lenguaje natural.")

    Boundary(taia, "Sistema TAIA", "Monolito modular") {
        Container(api, "API TAIA", "FastAPI", "Expone endpoints de Usuario, Academic, AI y Reminders.")
        Container(academic, "Módulo Academic", "Python", "Gestiona tareas académicas y aplica reglas de dominio.")
        Container(ai, "Módulo AI", "Python", "Gestiona conversaciones y traduce operaciones del asistente hacia los módulos de negocio.")
        Container(reminders, "Módulo Reminders", "Python", "Gestiona recordatorios y prepara/envía notificaciones.")
        Container(usuario, "Módulo Usuario", "Python", "Gestiona usuarios, JWT y vinculación de Telegram.")
        ContainerDb(memory, "Repositorios en memoria", "Python", "Persistencia temporal utilizada por la línea base ejecutable.")
    }

    Rel(estudiante, api, "Usa", "HTTPS/JSON")
    BiRel(telegram, api, "Vinculación y notificaciones", "HTTPS/JSON")
    Rel(api, usuario, "Autenticación e identidad")
    Rel(api, academic, "Operaciones académicas")
    Rel(api, ai, "Mensajes del asistente")
    Rel(api, reminders, "Gestión de recordatorios")
    Rel(ai, academic, "Consulta y modificación de tareas")
    Rel(reminders, academic, "Valida tareas asociadas")
    Rel(reminders, usuario, "Obtiene identidad/vinculación Telegram")
    Rel(ai, llm, "Interpreta mensajes", "HTTPS/JSON")
    Rel(reminders, telegram, "Envía notificaciones", "HTTPS/JSON")
    Rel(usuario, memory, "Lee/escribe")
    Rel(academic, memory, "Lee/escribe")
    Rel(reminders, memory, "Lee/escribe")

    UpdateElementStyle(estudiante, $bgColor="#08427b", $fontColor="#ffffff", $borderColor="#052e56")
    UpdateElementStyle(api, $bgColor="#438dd5", $fontColor="#ffffff", $borderColor="#2e6295")
```
