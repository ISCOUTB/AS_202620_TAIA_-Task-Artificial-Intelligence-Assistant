```mermaid
---
config:
  wrap: true
  c4:
    width: 240
---
C4Container
    title Diagrama de Contenedores - TAIA

    System_Ext(telegram, "Telegram", "Canal conversacional para capturar tareas y entregar notificaciones.")
    System_Ext(gemini, "Proveedor LLM ", "Interpreta el mensaje y extrae los campos estructurados de la tarea.")
    Person(estudiante, "Estudiante", "Registra tareas académicas y consulta su información.")

    Boundary(taia, "Sistema TAIA", "Sistema") {
        Container(appmovil, "App Móvil", "Consulta las tareas pendientes y permite registrarlas por formulario.")
        Container(api, "API TAIA", "Expone la API REST, recibe el webhook, aplica las reglas de negocio y programa recordatorios.")
        ContainerDb(db, "Base de Datos", "Almacena tareas, materias, horarios y usuarios.")
    }

    BiRel(estudiante, telegram, "Escribe y recibe avisos")
    Rel(estudiante, appmovil, "Consulta tareas")
    BiRel(telegram, api, "Mensajes", "JSON/HTTPS")
    BiRel(api, gemini, "Interpreta", "JSON/HTTPS")
    Rel(appmovil, api, "Consulta y registra", "JSON/HTTPS")
    Rel(api, db, "Lee y escribe", "SQL/TCP 5432")

    UpdateElementStyle(estudiante, $bgColor="#08427b", $fontColor="#ffffff", $borderColor="#052e56")
    UpdateElementStyle(appmovil, $bgColor="#438dd5", $fontColor="#ffffff", $borderColor="#2e6295")
    UpdateElementStyle(api, $bgColor="#438dd5", $fontColor="#ffffff", $borderColor="#2e6295")
    UpdateElementStyle(db, $bgColor="#438dd5", $fontColor="#ffffff", $borderColor="#2e6295")
    UpdateElementStyle(telegram, $bgColor="#999999", $fontColor="#ffffff", $borderColor="#6b6b6b")
    UpdateElementStyle(gemini, $bgColor="#999999", $fontColor="#ffffff", $borderColor="#6b6b6b")

    UpdateRelStyle(estudiante, telegram, $offsetX="-200", $offsetY="0")
    UpdateRelStyle(estudiante, appmovil, $offsetX="-30", $offsetY="-70")
    UpdateRelStyle(appmovil, api, $offsetX="-80", $offsetY="-85")

    UpdateLayoutConfig($c4ShapeInRow="2", $c4BoundaryInRow="1")
```

**Convenciones**

| Color | Significado |
| ----- | ----------- |
| Azul oscuro | Persona (actor humano) |
| Azul claro | Contenedor (unidad ejecutable o de almacenamiento) |
| Gris | Sistema externo (no lo construimos ni lo controlamos) |
